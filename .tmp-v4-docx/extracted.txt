   function calcStreak(logs: { brewedAt: Date }[]): number {
     if (!logs.length) return 0;
     const days = [...new Set(logs.map((l) => l.brewedAt.toISOString().slice(0, 10)))].sort().reverse();
     const today = new Date().toISOString().slice(0, 10);
     const yesterday = new Date(Date.now() - 86400000).toISOString().slice(0, 10);
     if (days[0] !== today && days[0] !== yesterday) return 0;
     let streak = 1;
     for (let i = 1; i < days.length; i++) {
       const prev = new Date(days[i - 1]);
       const curr = new Date(days[i]);
       if ((prev.getTime() - curr.getTime()) === 86400000) streak++;
       else break;
     }
     return streak;
   }

Prompt 13 â€” Keep screen awake during brew timer
Add screen wake lock to src/components/timer/brew-timer.tsx in Infusion Studio so the phone screen stays on while a brew is running.

1. Install @capacitor-community/keep-awake if not already in package.json:
   npm install @capacitor-community/keep-awake

2. In brew-timer.tsx, add a useEffect that watches isRunning:
   useEffect(() => {
     let wakeLock: WakeLockSentinel | null = null;
     async function acquireWakeLock() {
       try {
         if ('wakeLock' in navigator) {
           wakeLock = await (navigator as Navigator & { wakeLock: { request: (type: string) => Promise<WakeLockSentinel> } }).wakeLock.request('screen');
         } else {
           const { KeepAwake } = await import('@capacitor-community/keep-awake');
           await KeepAwake.keepAwake();
         }
       } catch { }
     }
     async function releaseWakeLock() {
       try {
         if (wakeLock) { await wakeLock.release(); wakeLock = null; }
         else {
           const { KeepAwake } = await import('@capacitor-community/keep-awake');
           await KeepAwake.allowSleep();
         }
       } catch { }
     }
     if (isRunning) acquireWakeLock();
     else releaseWakeLock();
     return () => { releaseWakeLock(); };
   }, [isRunning]);

The Web Wake Lock API is used on PWA/web, Capacitor KeepAwake is used as fallback for Android. Both are wrapped in try/catch so failure is silent.
